.data
input: .asciiz "-123"

.text
main: 
    la $a0, input //loads the address of the string
    jal checkSign

    move $a0, $v0
    li $v0, 1
    syscall //prints int result

    li $v0, 10
    syscall //exits program

checkSign: 
    lb $t0, 0($a0)
    li $t1, $0          //bool value for the sign (default is positive)
    bne $t0, "-", start
    addi $t1, $t1, 1    //bool value for the sign (now negative)
    addi $a0, $a0, 1    //next char
start:
    li $t3, 0           //sum = 0
loop:
    lb $t0, 0($a0)      //loads first char
    beqz $t0, bad       //check for null char
    blt $t0, "0", bad   //check for bad input
    bgt $t0, "9", bad   //check for bad input
    sub $t0, $t0, 48    //convert to ASCII
    mul $t3, $t3, 10    //mul the sum by 10
    add $t3, $t3, $t0   //add new char
    addi $a0, 1         //next char
    j loop
exit:
    beq $t1, 0, done
    neg $t3, $t3
done:
    move $v0, $t3
    jr $ra
bad:
    li $v0, -1
    jr $ra
