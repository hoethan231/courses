Polynomial Function:

.data
input .asciiz "Enter a value for x: "

.word
main:
    la $a0, input           //loads input as parameter
    li $v0, 4
    syscall                 //prints input
    
    li $v0, 5
    syscall                 //reads int
    move $a0, $v0           //moves the input into parameter for poly func

    jal poly                //calls poly function
    move $a0, $v0           //moves result to parameter for printing

    li $v0, 1
    syscall                 //prints result

    li $v0, 10
    syscall                 //exits program

poly:
    li $a1, 5               //sets the power to 5
    jal pow                 //calls pow function
    li $s0, 3
    mul $t0, $v0, $s0       //3x^5
    
    li $a1, 4           
    jal pow             
    li $s0, 2
    mul $t1, $v0, $s0       //2x^4
    add $t0, $t0, $t1       //(3x^5) + (2x^4)

    li $a1, 3
    jal pow             
    li $s0, 5
    mul $t1, $v0, $s0       //5x^3
    sub $t0, $t0, $t1       //(3x^5) + (2x^4) - (5x^3)

    li $a1, 2
    jal pow             
    sub $t0, $t0, $v0       //(3x^5) + (2x^4) - (5x^3) - x^2

    li $s0, 7
    mul $t1, $s0, $a0       //7x
    add $t0, $t0, $t1       //(3x^5) + (2x^4) - (5x^3) - x^2 + 7x

    addi $t0, $t0, -6       //(3x^5) + (2x^4) - (5x^3) - (x^2) + 7x - 6

    move $v0, $t0
    jr $ra                  //return polynomial

pow:
    li $s1, 1               //sum = 1
    li $s2, 1               //i = 1

loop:
    beq $s2, $a1, exit      //i == power
    mul $s1, $s1, $a0       //sum *= x
    addi $s2, $s2, 1        //i++

exit:
    move $v0, $s1
    jr $ra                  //return sum
